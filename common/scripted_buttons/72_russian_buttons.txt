##Vodka Monopoly Buttons

revoking_privileges_button = {
    name = "revoking_privileges_button"
    desc = "revoking_privileges_button_desc"

    visible = { always = yes }

    ai_chance = {
        value = 100
    }

    possible = {
        OR = {
            has_modifier = modifier_distillery_privileges
            has_modifier = modifier_monastic_distillery
        }
    }

    effect = {
        if = {
            limit = { has_modifier = modifier_distillery_privileges }
            remove_modifier = modifier_distillery_privileges
        }
        if = {
            limit = { has_modifier = modifier_monastic_distillery }
            remove_modifier = modifier_monastic_distillery
        }
        interest_group_landowners_approval_add = -5
    }
}
