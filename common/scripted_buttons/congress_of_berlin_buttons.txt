je_favour_russia_berlin_button = {
	name = "je_favour_russia_berlin_button"
	desc = "je_favour_russia_berlin_button_desc"

	visible = {
		NOT = {
			c:TUR ?= this
		}
	}

	possible = {
		NOR = {
			has_variable = favour_russia_berlin_country
			has_variable = favour_ottomans_berlin_country
			has_variable = favour_compromise_berlin_country
			has_global_variable = favour_berlin_bosnia_negotiation
			has_global_variable = favour_berlin_cyprus_negotiation
		}
	}

	ai_chance = {
		value = 100
	}

	effect = {
		set_variable = favour_russia_berlin_country
		change_global_variable = {
			name = russia_favour_berlin_progress
			add = ROOT.prestige
		}
	}
}
