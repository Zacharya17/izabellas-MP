bulgaria_negotiation.1 = {
	type = country_event
	placement = ROOT

	title = bulgaria_negotiation.1.t
	desc = bulgaria_negotiation.1.d # probably want a triggered desc here
	flavor = bulgaria_negotiation.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	immediate = {
	}
	option = {
		name = bulgaria_negotiation.1.a
		ai_chance = {
			base = 10
		}
		set_variable = favour_ottomans_berlin_country
		change_global_variable = {
			name = ottomans_favour_berlin_progress
			add = ROOT.prestige
		}
		remove_global_variable = favour_berlin_cyprus_negotiation
		s:STATE_CYPRUS = {
			random_scope_state = {
				limit = {
					exists = c:GBR
					owner = {
						OR = {
							c:TUR = THIS
						}
					}
				}
			set_state_owner = c:GBR
			}
		}
	}

	option = {
		name = bulgaria_negotiation.1.b
		remove_global_variable = favour_berlin_cyprus_negotiation
	}
}
